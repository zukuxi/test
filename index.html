async function getPage(url) { const target = new URL(url); if (target.protocol !== "https:" && target.protocol !== "http:") { throw new Error("只支持 HTTP 或 HTTPS 网址。"); } // corsproxy.nl 使用路径形式转发目标网址 const proxyUrl = "https://corsproxy.nl/" + target.protocol.slice(0, -1) + "/" + target.host + target.pathname + target.search + target.hash; const response = await fetch(proxyUrl, { method: "GET", cache: "no-store" }); const responseText = await response.text(); if (!response.ok) { throw new Error( "代理请求失败，HTTP " + response.status + "\n代理返回：" + responseText.slice(0, 300) ); } if (!responseText || responseText.length < 100) { throw new Error("返回内容为空或过短，可能是代理或目标网站拦截。"); } return new DOMParser().parseFromString(responseText, "text/html"); }
